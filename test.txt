import { useEffect, useRef } from 'react';
import { io, type Socket } from 'socket.io-client';
import {
  getAccessToken,
  getApiBaseUrl,
  refreshPortalSession,
  type Role,
  type TenantIds,
} from '../api/http';
import type { ApiNotification } from '../types';

type RealtimeHandlers = {
  onNotification?: (item: ApiNotification) => void;
  onStaffActivity?: (payload: Record<string, unknown>) => void;
  onBookingProgress?: (payload: Record<string, unknown>) => void;
  onConnectChange?: (connected: boolean) => void;
};

const ENABLE_WEBSOCKET = String(import.meta.env.VITE_SOCKET_WEBSOCKET || '').toLowerCase() === 'true';

let sharedSocket: Socket | null = null;
let sharedKey = '';

function socketKey(role: Role, token: string, tenant: TenantIds | null | undefined): string {
  const t = tenant ? `${tenant.clientId}|${tenant.regionId}|${tenant.storeId}` : '';
  return `${role}:${token.slice(0, 12)}:${t}`;
}

function disconnectShared(): void {
  if (sharedSocket) {
    sharedSocket.removeAllListeners();
    sharedSocket.disconnect();
    sharedSocket = null;
  }
  sharedKey = '';
}

/**
 * Maintains one Socket.IO connection per active portal session (customer / groomer / staff).
 */
export function useRealtimeSocket(
  enabled: boolean,
  role: Role,
  tenant: TenantIds | null | undefined,
  handlers: RealtimeHandlers
): void {
  const handlersRef = useRef(handlers);
  handlersRef.current = handlers;

  useEffect(() => {
    if (!enabled) {
      disconnectShared();
      return undefined;
    }

    const token = getAccessToken(role);
    if (!token) {
      disconnectShared();
      return undefined;
    }

    const key = socketKey(role, token, tenant);
    if (sharedSocket && sharedKey === key && sharedSocket.connected) {
      return undefined;
    }

    disconnectShared();
    sharedKey = key;

    const apiBase = getApiBaseUrl();
    const tenantPayload = tenant || undefined;
    const socket = io(apiBase || undefined, {
      path: '/socket.io',
      // Function form: re-read the latest access token on every (re)connect attempt,
      // so a token refreshed by the HTTP client is picked up automatically.
      auth: (cb) => cb({ token: getAccessToken(role) ?? '', tenant: tenantPayload }),
      // The production proxy currently blocks the wss:// upgrade, so stay on HTTP long-polling.
      // Set VITE_SOCKET_WEBSOCKET=true once the proxy forwards Upgrade/Connection headers.
      transports: ENABLE_WEBSOCKET ? ['polling', 'websocket'] : ['polling'],
      reconnection: true,
      reconnectionAttempts: 10,
    });
    sharedSocket = socket;

    const reportConnect = (connected: boolean) => {
      handlersRef.current.onConnectChange?.(connected);
    };

    // Auth failures from the server middleware stop Socket.IO's auto-reconnect,
    // so refresh the session once and reconnect manually.
    let authRetries = 0;
    const MAX_AUTH_RETRIES = 1;

    socket.on('connect', () => {
      authRetries = 0;
      reportConnect(true);
    });
    socket.on('disconnect', (reason) => {
      if (import.meta.env.DEV) console.info('[socket] disconnected:', reason);
      reportConnect(false);
    });
    socket.on('connect_error', async (err: Error) => {
      console.error('[socket] connect_error:', err.message);
      reportConnect(false);

      // socket.active === false means the server rejected the handshake (middleware error);
      // transport errors keep active === true and are retried by the built-in reconnection.
      const isAuthError = /token|unauthori[sz]ed|jwt|expired|forbidden/i.test(err.message);
      if (socket.active || !isAuthError || authRetries >= MAX_AUTH_RETRIES) return;

      authRetries += 1;
      const refreshed = await refreshPortalSession(role, apiBase).catch(() => false);
      if (refreshed && sharedSocket === socket && getAccessToken(role)) {
        socket.connect();
      }
    });
    reportConnect(socket.connected);

    socket.on('notification', (raw: ApiNotification) => {
      if (!raw?.title) return;
      handlersRef.current.onNotification?.({
        id: raw.id ?? 0,
        title: raw.title,
        message: raw.message,
        type: raw.type || 'general',
        data: raw.data || {},
        isRead: Boolean(raw.isRead),
        createdAt: raw.createdAt || new Date().toISOString(),
      });
    });

    socket.on('staff_activity', (payload: Record<string, unknown>) => {
      handlersRef.current.onStaffActivity?.(payload);
    });

    socket.on('booking_progress', (payload: Record<string, unknown>) => {
      handlersRef.current.onBookingProgress?.(payload);
    });

    return () => {
      disconnectShared();
    };
  }, [enabled, role, tenant?.clientId, tenant?.regionId, tenant?.storeId]);
}
